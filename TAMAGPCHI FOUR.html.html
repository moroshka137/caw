function checkAchievements(){
  const user = Auth.currentUser(); if(!user) return;
  const unlocked = user.achievements || [];
  const newOnes = [];
  ACHIEVEMENTS.forEach(a => {
    if(unlocked.includes(a.id)) return;
    if(a.check(user)){ unlocked.push(a.id); newOnes.push(a); }
  });

    Auth.updateUser(user.login, { achievements: unlocked });
    newOnes.forEach((a, i) => setTimeout(() => showToast(a), i*800));
  }
}

}

function getEmotion(pet){
  const h = new Date().getHours();
  if(h >= 22 || h < 6) return { id:'sleepy', icon:'😴', label:'Сонный', cls:'emotion-sleepy', phrase:'Zzz...' };
  if(pet.hunger < 25) return { id:'hungry', icon:'😫', label:'Голодный', cls:'emotion-hungry', phrase:'Я голоден!' };
  if(pet.mood < 30)   return { id:'sad',    icon:'😢', label:'Грустный', cls:'emotion-sad',    phrase:'Мне грустно...' };
  if(pet.hunger > 80 && pet.mood > 80) return { id:'happy', icon:'🤩', label:'Счастливый', cls:'emotion-happy', phrase:'Ура! Я счастлив!' };
  return { id:'normal', icon:'🙂', label:'Обычный', cls:'', phrase:'' };
}

function getTimeClass(){
  const h = new Date().getHours();
  if(h >= 6 && h < 12) return 'time-morning';
  if(h >= 12 && h < 18) return 'time-day';
  if(h >= 18 && h < 22) return 'time-evening';
  return 'time-night';
}

  function PacmanGame({ canvas, ctx, title, hud, actions, help, onFinish }){
    const TILE=28, COLS=21, ROWS=17;
    const W=COLS*TILE, H=ROWS*TILE;
    canvas.width=W; canvas.height=H;
    const MAP=[
      "#####################","#.........#.........#","#.###.###.#.###.###.#","#.#...........#.....#",
      "#.#.###.#####.###.#.#","#.....#...#...#.....#","#####.#.#.#.#.#.#####","    #.#.#.#.#.#.#    ",
      "#####.#.#.#.#.#.#####","#.........#.........#","#.###.###.#.###.###.#","#.#...............#.#",
      "#.#.#####.#.#####.#.#","#.....#...#...#.....#","#.###.#.#####.#.###.#","#.........#.........#","#####################"
    ];
    let pac, ghosts, dots, totalDots, eaten, state, tick, keys, score, timeLeft;
    function reset(){
      pac = { x:9.5*TILE, y:12.5*TILE, r:TILE*0.4, dx:-1, dy:0, nextDx:-1, nextDy:0, speed:2.1 };
      ghosts = [
        { x:9.5*TILE,y:6.5*TILE,dx:0,dy:-1,color:'#ff4d4d',speed:1.5 },
        { x:8.5*TILE,y:6.5*TILE,dx:0,dy:-1,color:'#ff9ecb',speed:1.4 },
        { x:10.5*TILE,y:6.5*TILE,dx:0,dy:-1,color:'#4dd2ff',speed:1.4 }
      ];
      dots=[];
      for(let r=0;r<ROWS;r++) for(let c=0;c<COLS;c++) if(MAP[r][c]==='.') dots.push({c,r,got:false});
      totalDots=dots.length; eaten=0; score=0; timeLeft=60; state=true; tick=0; keys={};
      document.onkeydown = e => { keys[e.key]=true; if(['ArrowUp','ArrowDown','ArrowLeft','ArrowRight'].includes(e.key)) e.preventDefault(); };
      document.onkeyup = e => { keys[e.key]=false; };
    }
    function isWall(px,py){ const c=Math.floor(px/TILE), r=Math.floor(py/TILE); if(r<0||r>=ROWS||c<0||c>=COLS) return true; return MAP[r][c]==='#'; }
    function canMove(x,y){ const r=TILE*0.35; return !isWall(x-r,y-r)&&!isWall(x+r,y-r)&&!isWall(x-r,y+r)&&!isWall(x+r,y+r); }
    function movePac(){
      if(keys['ArrowLeft']||keys['a']||keys['A']) { pac.nextDx=-1; pac.nextDy=0; }
      if(keys['ArrowRight']||keys['d']||keys['D']){ pac.nextDx=1; pac.nextDy=0; }
      if(keys['ArrowUp']||keys['w']||keys['W']){ pac.nextDx=0; pac.nextDy=-1; }
      if(keys['ArrowDown']||keys['s']||keys['S']){ pac.nextDx=0; pac.nextDy=1; }
      const nx0 = pac.x + pac.nextDx*pac.speed, ny0 = pac.y + pac.nextDy*pac.speed;
      if((pac.nextDx!==0||pac.nextDy!==0) && canMove(nx0,ny0)){ pac.dx=pac.nextDx; pac.dy=pac.nextDy; }
      const nx = pac.x + pac.dx*pac.speed, ny = pac.y + pac.dy*pac.speed;
      if(canMove(nx,ny)){ pac.x=nx; pac.y=ny; }
    }
    function eatDots(){
      dots.forEach(d => {
        if(d.got) return;
        const dx=(d.c+.5)*TILE-pac.x, dy=(d.r+.5)*TILE-pac.y;
        if(Math.hypot(dx,dy)<pac.r+4){
          d.got=true; eaten++; score+=10; Sounds.play('coin');
          if(eaten===totalDots){ Auth.addStat('pacmanWins'); done(`Победа! 🎉`, Math.min(200, score/2 + timeLeft*2)); }
        }
      });
    }
    function moveGhosts(){
      ghosts.forEach(g => {
        const dirs=[{dx:1,dy:0},{dx:-1,dy:0},{dx:0,dy:1},{dx:0,dy:-1}];
        dirs.sort((a,b) => {
          const ax=g.x+a.dx*TILE, ay=g.y+a.dy*TILE;
          const bx=g.x+b.dx*TILE, by=g.y+b.dy*TILE;
          return Math.hypot(ax-pac.x,ay-pac.y)-Math.hypot(bx-pac.x,by-pac.y);
        });
        const ok = dirs.filter(d => canMove(g.x+d.dx*g.speed, g.y+d.dy*g.speed) && !(d.dx===-g.dx && d.dy===-g.dy));
        const choose = ok[Math.random()<0.85?0:Math.floor(Math.random()*ok.length)] || dirs[0];
        const nx=g.x+choose.dx*g.speed, ny=g.y+choose.dy*g.speed;
        if(canMove(nx,ny)){ g.x=nx; g.y=ny; g.dx=choose.dx; g.dy=choose.dy; }
        if(Math.hypot(g.x-pac.x,g.y-pac.y)<TILE*0.7){ done('Поймали! 💀', 0); }
      });
    }
    function done(text, earned){
      if(!state) return; state=false; document.onkeydown=null; document.onkeyup=null;
      onFinish(earned, text);
    }
    function update(){
      if(!state) return; tick++;
      if(tick%60===0){ timeLeft--; if(timeLeft<=0){ done('Время вышло! ⏰', 0); return; } }
      movePac(); eatDots(); moveGhosts();
      hud.innerHTML = `<div>🎯 Очки: ${score}</div><div>⚪ Точки: ${eaten}/${totalDots}</div><div>⏱ ${timeLeft} сек</div>`;
    }
    function draw(){
      ctx.clearRect(0,0,W,H);
      ctx.fillStyle='#0a0a1a'; ctx.fillRect(0,0,W,H);
      for(let r=0;r<ROWS;r++) for(let c=0;c<COLS;c++){
        if(MAP[r][c]==='#'){
          ctx.fillStyle='#1a1a5e'; ctx.fillRect(c*TILE,r*TILE,TILE,TILE);
          ctx.strokeStyle='#4a4aff'; ctx.lineWidth=2; ctx.strokeRect(c*TILE+2,r*TILE+2,TILE-4,TILE-4);
        }
      }
      dots.forEach(d => { if(d.got) return; ctx.fillStyle='#ffe066'; ctx.beginPath(); ctx.arc((d.c+.5)*TILE,(d.r+.5)*TILE,3,0,7); ctx.fill(); });
      ghosts.forEach(g => {
        ctx.fillStyle=g.color;
        ctx.beginPath(); ctx.arc(g.x,g.y,TILE*0.4,Math.PI,0);
        ctx.lineTo(g.x+TILE*0.4,g.y+TILE*0.4); ctx.lineTo(g.x+TILE*0.2,g.y+TILE*0.25); ctx.lineTo(g.x,g.y+TILE*0.4);
        ctx.lineTo(g.x-TILE*0.2,g.y+TILE*0.25); ctx.lineTo(g.x-TILE*0.4,g.y+TILE*0.4);
        ctx.fill();
        ctx.fillStyle='#fff'; ctx.beginPath(); ctx.arc(g.x-5,g.y-4,4,0,7); ctx.arc(g.x+5,g.y-4,4,0,7); ctx.fill();
        ctx.fillStyle='#1a1a1a'; ctx.beginPath(); ctx.arc(g.x-5+g.dx*2,g.y-4+g.dy*2,2,0,7); ctx.arc(g.x+5+g.dx*2,g.y-4+g.dy*2,2,0,7); ctx.fill();
      });
      const mouth=Math.abs(Math.sin(tick*0.15))*0.5;
      const angle=Math.atan2(pac.dy,pac.dx);
      ctx.fillStyle='#ffe600'; ctx.beginPath(); ctx.moveTo(pac.x,pac.y);
      ctx.arc(pac.x,pac.y,pac.r,angle+mouth,angle-mouth+Math.PI*2); ctx.closePath(); ctx.fill();
      ctx.fillStyle='#1a1a1a';
      const ex=pac.x+Math.cos(angle-Math.PI/2)*pac.r*0.4, ey=pac.y+Math.sin(angle-Math.PI/2)*pac.r*0.4;
      ctx.beginPath(); ctx.arc(ex,ey,2.5,0,7); ctx.fill();
    }
    function loop(){ if(!state) return; update(); draw(); requestAnimationFrame(loop); }
    return {
      start(){ title.textContent='👻 Пак-пак'; help.textContent='← → ↑ ↓ или WASD. Съешь все точки и не дай призракам поймать!'; actions.innerHTML=`<button class="btn btn-soft" onclick="App.go('minigames')">← Назад</button>`; reset(); loop(); },
      stop(){ state=false; document.onkeydown=null; document.onkeyup=null; }
    };
  }

  function RaceGame({ canvas, ctx, title, hud, actions, help, onFinish }){
    const W=canvas.width, H=canvas.height;
    const ROAD_LEFT=W*0.15, ROAD_RIGHT=W*0.85;
    let player, obstacles, bonuses, speed, distance, keys, state, tick, gameTime;
    function reset(){
      player={ x:W/2, y:H-90, w:46, h:80 };
      obstacles=[]; bonuses=[]; speed=3.2; distance=0; tick=0; gameTime=45; keys={}; state=true;
      document.onkeydown = e => { keys[e.key]=true; if(['ArrowLeft','ArrowRight',' '].includes(e.key)) e.preventDefault(); };
      document.onkeyup = e => { keys[e.key]=false; };
      actions.innerHTML = `
        <button class="btn btn-soft" id="leftBtn">◀</button>
        <button class="btn btn-soft" id="rightBtn">▶</button>
        <button class="btn btn-primary" id="boostBtn">⚡</button>
        <button class="btn btn-soft" onclick="App.go('minigames')">← Назад</button>
      `;
      const press = (key, el) => { el.onpointerdown = () => keys[key]=true; el.onpointerup = () => keys[key]=false; el.onpointerleave = () => keys[key]=false; };
      press('ArrowLeft', document.getElementById('leftBtn'));
      press('ArrowRight', document.getElementById('rightBtn'));
      press(' ', document.getElementById('boostBtn'));
    }
    function update(){
      if(!state) return; tick++;
      if(tick%60===0){ gameTime--; if(gameTime<=0){ finish(distance); return; } }
      const ms = 6;
      if(keys['ArrowLeft']||keys['a']||keys['A']) player.x-=ms;
      if(keys['ArrowRight']||keys['d']||keys['D']) player.x+=ms;
      player.x = Math.max(ROAD_LEFT+player.w/2, Math.min(ROAD_RIGHT-player.w/2, player.x));
      const boosting = keys[' ']||keys['Shift'];
      const base = 3.2 + distance/1500;
      speed = boosting ? base*1.7 : base;
      distance += speed;
      if(tick % Math.max(18, 40-Math.floor(distance/200)) === 0){
        const type = Math.random()<0.7 ? 'cone' : 'car';
        obstacles.push({ x:ROAD_LEFT+Math.random()*(ROAD_RIGHT-ROAD_LEFT-50)+25, y:-60, w:46, h:type==='car'?80:40, type,
          color:type==='car'?['#e63946','#4d9eff','#2ecc71','#f39c12'][Math.floor(Math.random()*4)]:'#ff8c42' });
      }
      if(tick%90===0) bonuses.push({ x:ROAD_LEFT+Math.random()*(ROAD_RIGHT-ROAD_LEFT-30)+15, y:-30, r:14, got:false });
      obstacles.forEach(o => o.y += speed*1.4);
      obstacles = obstacles.filter(o => o.y < H+100);
      bonuses.forEach(b => b.y += speed*1.4);
      bonuses = bonuses.filter(b => b.y < H+50);
      for(const o of obstacles){
        if(player.x+player.w/2>o.x-o.w/2 && player.x-player.w/2<o.x+o.w/2 && player.y+player.h/2>o.y-o.h/2 && player.y-player.h/2<o.y+o.h/2){ finish(distance, true); return; }
      }
      bonuses.forEach(b => { if(!b.got && Math.hypot(player.x-b.x,player.y-b.y)<30){ b.got=true; distance+=200; Sounds.play('coin'); } });
      hud.innerHTML = `<div>🛣️ ${Math.floor(distance/10)} м</div><div>⏱ ${gameTime} сек</div><div>⚡ ${Math.round(speed*10)}</div>`;
    }
    function drawRoad(){
      ctx.fillStyle='#2e7d32'; ctx.fillRect(0,0,ROAD_LEFT,H); ctx.fillRect(ROAD_RIGHT,0,W,H);
      ctx.fillStyle='#3a3a3a'; ctx.fillRect(ROAD_LEFT,0,ROAD_RIGHT-ROAD_LEFT,H);
      ctx.fillStyle='#f5e6a3';
      const dash=60, offset=(distance*1.4)%(dash*2);
      for(let y=-dash*2+offset; y<H; y+=dash*2){
        ctx.fillRect(W/2-3,y,6,dash); ctx.fillRect(ROAD_LEFT+10,y,4,dash); ctx.fillRect(ROAD_RIGHT-14,y,4,dash);
      }
    }
    function drawPlayerCar(){
      const x=player.x, y=player.y, w=player.w, h=player.h;
      ctx.fillStyle='rgba(0,0,0,.35)'; ctx.beginPath(); ctx.ellipse(x,y+h/2,w/2*1.1,h/3,0,0,7); ctx.fill();
      ctx.fillStyle='#e63946'; ctx.beginPath(); ctx.roundRect(x-w/2,y-h/2,w,h,10); ctx.fill();
      ctx.fillStyle='#b3202c'; ctx.beginPath(); ctx.roundRect(x-w/2+4,y-h/2+6,w-8,h*0.35,6); ctx.fill();
      ctx.fillStyle='#a5d8ff'; ctx.beginPath(); ctx.roundRect(x-w/2+6,y-h/4,w-12,h*0.28,4); ctx.fill();
      ctx.fillStyle='#111';
      ctx.fillRect(x-w/2-4,y-h/3,8,16); ctx.fillRect(x+w/2-4,y-h/3,8,16);
      ctx.fillRect(x-w/2-4,y+h/3-16,8,16); ctx.fillRect(x+w/2-4,y+h/3-16,8,16);
      ctx.fillStyle='#fff8b0'; ctx.beginPath(); ctx.arc(x-w/4,y-h/2+4,4,0,7); ctx.arc(x+w/4,y-h/2+4,4,0,7); ctx.fill();
    }
    function drawObstacles(){
      obstacles.forEach(o => {
        if(o.type==='cone'){
          ctx.fillStyle='#ff8c42'; ctx.beginPath(); ctx.moveTo(o.x,o.y-o.h/2); ctx.lineTo(o.x-o.w/2,o.y+o.h/2); ctx.lineTo(o.x+o.w/2,o.y+o.h/2); ctx.closePath(); ctx.fill();
          ctx.fillStyle='#fff'; ctx.fillRect(o.x-6,o.y-2,12,6);
        } else {
          ctx.fillStyle=o.color; ctx.beginPath(); ctx.roundRect(o.x-o.w/2,o.y-o.h/2,o.w,o.h,8); ctx.fill();
          ctx.fillStyle='rgba(255,255,255,.4)'; ctx.beginPath(); ctx.roundRect(o.x-o.w/2+5,o.y-o.h/3,o.w-10,o.h*0.3,4); ctx.fill();
        }
      });
    }
    function drawBonuses(){
      bonuses.forEach(b => { if(b.got) return; ctx.fillStyle='#ffd93d'; ctx.beginPath(); ctx.arc(b.x,b.y,b.r,0,7); ctx.fill(); ctx.strokeStyle='#d4a017'; ctx.lineWidth=2; ctx.stroke(); });
    }
    function draw(){ ctx.clearRect(0,0,W,H); drawRoad(); drawObstacles(); drawBonuses(); drawPlayerCar(); }
    function finish(dist, crash){
      if(!state) return; state=false;
      document.onkeydown=null; document.onkeyup=null;
      const earned = crash ? Math.floor(dist/30) : Math.min(250, Math.floor(dist/15));
      if(!crash) Auth.addStat('raceWins', earned);
      onFinish(earned, crash ? `💥 Авария! Дистанция: ${Math.floor(dist/10)} м.` : `Финиш! Дистанция: ${Math.floor(dist/10)} м.`);
    }
    function loop(){ if(!state) return; update(); draw(); requestAnimationFrame(loop); }
    return {
      start(){ title.textContent='🏎️ Гонки'; help.textContent='← → двигаться, Пробел — ускорение. Уворачивайся!'; reset(); loop(); },
      stop(){ state=false; document.onkeydown=null; document.onkeyup=null; }
    };
  }

        hunger: pendingNameSpecies.hunger,
        mood: pendingNameSpecies.mood,
        diary: [],
        lastDiaryDate: null,
        createdAt: Date.now()
      };
      Auth.updateUser(user.login, { pets: [...(user.pets||[]), newPet], activePetUid: newPet.uid });
      Auth.addStat('petsCount');
      closeModal('namePetModal');
      Sounds.play('happy');
      pendingNameSpecies=null;
      checkAchievements();
      go('room');
    };

    document.getElementById('buyPetCancel').onclick = () => { closeModal('buyPetModal'); pendingBuySpecies=null; };
    document.getElementById('buyPetConfirm').onclick = () => {
      if(!pendingBuySpecies) return;
      const user = Auth.currentUser();
      if(user.coins < pendingBuySpecies.price){ alert('💸 Не хватает монет!'); return; }
      Auth.updateUser(user.login, { coins: user.coins - pendingBuySpecies.price });
      closeModal('buyPetModal');
      pendingNameSpecies = pendingBuySpecies;
      pendingBuySpecies = null;
      document.getElementById('namePetImg').textContent = pendingNameSpecies.emoji;
      document.getElementById('namePetInput').value = pendingNameSpecies.defaultName;
      openModal('namePetModal');
      Sounds.play('coin');
    };
  }

  function init(){
    initForms();
    renderHeader();
    renderHome();

    setInterval(() => {
      const user = Auth.currentUser(); if(!user || !user.pets || user.pets.length===0) return;
      let changed = false;
      user.pets.forEach(p => {
        if(p.hunger > 0){ p.hunger = Math.max(0, p.hunger-1); changed = true; }
        if(p.mood > 0 && p.hunger > 30){ p.mood = Math.max(0, p.mood-1); changed = true; }
      });
      if(changed){
        Auth.updateUser(user.login, { pets: user.pets });
        if(currentScreen === 'room')  renderRoom();
        if(currentScreen === 'rooms') renderHotel();
      }
    }, 60000);

    setInterval(() => {
      if(currentScreen === 'room'){
        const room = document.getElementById('roomStage'); if(!room) return;
        room.classList.remove('time-morning','time-day','time-evening','time-night');
        room.classList.add(getTimeClass());
      }
    }, 5*60*1000);
  }

  return { go, init, toggleSound, startGame };
})();
